\chapter{Temporal graphs}
\label{chap:model}

This chapter fixes the basic model: a temporal graph is a relation labeled by
discrete time, and we recall the handful of predicates on such a relation
(symmetry, looplessness, simplicity, being a clique) that the rest of the
library builds on. Formalized in \texttt{TemporalGraph/Basic.lean}; the
bipartite section (Section~\ref{sec:bipartite}: bipartiteness, restriction to the arcs
between two sets, bi-cliques) is in \texttt{TemporalGraph/Bipartite.lean}.

\begin{definition}[Temporal graph]
  \label{def:temporal-graph}
  \lean{TemporalGraph}
  \leanok
  A \emph{temporal graph} on a vertex type $V$ is a relation
  $\mathrm{Adj} : V \to V \to \mathbb{N} \to \mathrm{Prop}$, where
  $\mathrm{Adj}(x,y,t)$ means there is an edge between $x$ and $y$ present at
  time $t$. No symmetry or irreflexivity is assumed a priori; see
  Definition~\ref{def:is-symm-loopless}.
\end{definition}

\begin{definition}[Symmetric, loopless]
  \label{def:is-symm-loopless}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.IsSymm}
  \lean{TemporalGraph.IsLoopless}
  \leanok
  A temporal graph $G$ \emph{is symmetric} if $\mathrm{Adj}_G(x,y,t) \iff
  \mathrm{Adj}_G(y,x,t)$ for every $x,y,t$; it \emph{is loopless} if
  $\mathrm{Adj}_G(x,x,t)$ never holds.
\end{definition}

\begin{definition}[Simple, clique]
  \label{def:is-simple-clique}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.IsSimple}
  \lean{TemporalGraph.IsClique}
  \lean{TemporalGraph.IsCliqueGraph}
  \leanok
  $G$ \emph{is simple} if each ordered pair of vertices is joined at most
  at one time: there is at most one time $n$ with $\mathrm{Adj}_G(x,y,n)$
  (i.e.\ $\mathrm{Adj}_G(x,y,n) \wedge \mathrm{Adj}_G(x,y,m)
  \Rightarrow n = m$). A set $S$ of vertices \emph{forms a clique} in $G$ if
  every pair of distinct vertices in $S$ is joined by an edge of $G$ at some
  time; $G$ \emph{is a clique graph} (\texttt{IsCliqueGraph}) when $S$ is the set of all vertices.
\end{definition}

\begin{definition}[Footprint]
  \label{def:footprint}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.footprint}
  \leanok
  The \emph{footprint} of $G$ is the (time-forgetting) directed graph on $V$
  obtained by keeping an arc $x \to y$ whenever $x$ and $y$ are adjacent at
  some time. It is an ordinary (non-temporal) directed graph, a Mathlib
  \texttt{Digraph}.
\end{definition}

\begin{lemma}[Clique iff complete footprint]
  \label{lem:isClique-iff-footprint-complete}
  \uses{def:is-simple-clique,def:footprint}
  \lean{TemporalGraph.isClique_iff_footprint_isComplete}
  \leanok
  A set $S$ of vertices forms a clique in $G$ exactly when the vertices of $S$
  are pairwise adjacent in $G$'s footprint (i.e.\ the footprint restricted to
  $S$ is complete).
\end{lemma}

\begin{proof}
  \leanok
  Direct from the definitions: both sides say that distinct vertices of $S$ are joined by an arc at some time.
\end{proof}

\begin{definition}[Proper, happy]
  \label{def:is-proper-happy}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.IsProper}
  \lean{TemporalGraph.IsHappy}
  \leanok
  $G$ \emph{is proper} if, at any given time, a vertex has at most one
  out-neighbor (i.e.\ $\mathrm{Adj}_G(x,y,n) \wedge \mathrm{Adj}_G(x,z,n)
  \Rightarrow y = z$). $G$ \emph{is happy} if it is both simple
  (Definition~\ref{def:is-simple-clique}) and proper: between any ordered
  pair there is at most one time of adjacency, and at any given time a
  vertex has at most one out-neighbor.
\end{definition}

\begin{definition}[Earliest / latest neighbor]
  \label{def:earliest-latest-neighbor}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.EarliestEdge}
  \lean{TemporalGraph.LatestEdge}
  \lean{TemporalGraph.EarliestNeighbor}
  \lean{TemporalGraph.LatestNeighbor}
  \leanok
  For $x \in V$, an edge $(x,y,n)$ is the \emph{earliest edge} of $x$ towards
  $y$ if $n$ is minimal among times at which $x$ has an edge; it is the
  \emph{latest edge} if $n$ is maximal. $y$ is an \emph{earliest} (resp.\
  \emph{latest}) \emph{neighbor} of $x$ if such an edge exists. These notions
  drive the dismountability argument of Chapter~\ref{chap:dismountability}.
\end{definition}

\begin{definition}[Incidence set]
  \label{def:incidence-set}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.incidenceSet}
  \leanok
  The \emph{incidence set} of $x$ in $G$ (a temporal graph, or a
  sub-temporal-graph through inheritance, Chapter~\ref{chap:spanners}) is the
  set of pairs $(y,n)$ such that $\mathrm{Adj}_G(x,y,n)$: $x$'s out-neighbors,
  each paired with a time witnessing the edge.
\end{definition}

\begin{definition}[Spanning sub-temporal-graph order]
  \label{def:spanning-order}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.IsSpanningSubgraph}
  \leanok
  $G$ is a \emph{spanning sub-temporal-graph} of $H$ (written $G \le H$) if
  every edge of $G$ is an edge of $H$: $\mathrm{Adj}_G(x,y,n) \Rightarrow
  \mathrm{Adj}_H(x,y,n)$ for all $x,y,n$.
\end{definition}

\begin{lemma}[Monotonicity of simple, proper, happy]
  \label{lem:monotone-simple-proper-happy}
  \uses{def:spanning-order,def:is-simple-clique,def:is-proper-happy}
  \lean{TemporalGraph.IsSimple.mono}
  \lean{TemporalGraph.IsProper.mono}
  \lean{TemporalGraph.IsHappy.mono}
  \leanok
  If $H$ is simple (resp.\ proper, resp.\ happy) and $G \le H$
  (Definition~\ref{def:spanning-order}), then $G$ is simple (resp.\ proper,
  resp.\ happy): each of these predicates only restricts the edges that are
  present, so it is inherited by any spanning sub-temporal-graph.
\end{lemma}

\begin{proof}
  \leanok
  An arc of $G$ is an arc of $H$, so two arcs of $G$ violating simplicity (resp.\ properness) would violate it in $H$. Happiness is the conjunction of the two.
\end{proof}

\begin{definition}[General sub-temporal-graph]
  \label{def:is-subgraph-of}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.IsSubgraphOf}
  \leanok
  Let $P, P'$ be predicates on $V$. A temporal graph $G$ carried by the
  vertices satisfying $P$ is a \emph{sub-temporal-graph} of $H$, carried by
  those satisfying $P'$, when $P$ implies $P'$ and every edge of $G$ is an
  edge of $H$ between the corresponding vertices. This is the general
  inclusion: it compares graphs living on two different vertex subsets, and
  Definition~\ref{def:spanning-order} is its special case $P = P'$.
\end{definition}

\begin{definition}[Arcs and edges]
  \label{def:arcset-edgeset}
  \uses{def:temporal-graph,def:is-symm-loopless}
  \lean{TemporalGraph.arcSet}
  \lean{TemporalGraph.edgeSet}
  \lean{TemporalGraph.mem_edgeSet_iff_or}
  \lean{TemporalGraph.arcSize}
  \lean{TemporalGraph.size}
  \lean{TemporalGraph.size_eq_ncard}
  \lean{TemporalGraph.arcSize_eq_ncard}
  \leanok
  The \emph{arcs} of $G$ are the triples $(x,y,n)$ with $\mathrm{Adj}_G(x,y,n)$:
  an oriented notion, counted by $\mathrm{arcSize}$. The
  \emph{edges} of $G$ are unordered: pairs $(\{x,y\}, n)$ such that
  $\mathrm{Adj}_G(x,y,n)$ or $\mathrm{Adj}_G(y,x,n)$, counted by
  $\mathrm{size}$. An unordered pair is therefore an edge exactly when one of
  its orientations is an arc, with no hypothesis; when $G$ is symmetric the
  two orientations coincide and membership reduces to
  $\mathrm{Adj}_G(x,y,n)$. Both $\mathrm{size}$ and $\mathrm{arcSize}$ are
  computable \texttt{Fintype.card} of the edge/arc set, under a
  \texttt{Fintype} instance (the same pattern as Mathlib's
  \texttt{SimpleGraph.degree}); the instance-free \texttt{Set.ncard} form is
  always available via $\mathrm{size\_eq\_ncard}$ (an infinite edge or arc set
  then counts as $0$).
\end{definition}

\begin{definition}[Symmetrization]
  \label{def:symmetrize}
  \uses{def:temporal-graph,def:is-symm-loopless,def:spanning-order}
  \lean{TemporalGraph.symmetrize}
  \lean{TemporalGraph.le_symmetrize}
  \lean{TemporalGraph.symmetrize_le}
  \leanok
  The \emph{symmetrization} of $G$ has $\mathrm{Adj}(x,y,n)$ whenever
  $\mathrm{Adj}_G(x,y,n)$ or $\mathrm{Adj}_G(y,x,n)$. It is symmetric, it lies
  above $G$, and it lies below every symmetric graph above $G$ — that is, it is
  the least symmetric temporal graph above $G$. It adds no loop, so it
  preserves looplessness.
\end{definition}

\begin{definition}[Building a temporal graph]
  \label{def:from-arc-edge-set}
  \uses{def:arcset-edgeset}
  \lean{TemporalGraph.fromArcSet}
  \lean{TemporalGraph.fromEdgeSet}
  \leanok
  A set of arcs yields a temporal graph directly, with no symmetry assumed. A
  set of unordered edges yields one that is symmetric by construction, each
  pair being read in both directions; its edge set is the set one started
  from.
\end{definition}

\section{Bipartite temporal graphs and bi-cliques}
\label{sec:bipartite}

\begin{definition}[Bipartiteness]
  \label{def:is-bipartite}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.IsBipartiteWith}
  \lean{TemporalGraph.IsBipartite}
  \leanok
  A temporal graph $G$ \emph{is bipartite} relative to a pair of disjoint sets
  $(s, t)$ of vertices if every edge of $G$ connects a vertex in $s$ to a
  vertex in $t$ (in either direction, with no symmetry required). The
  \emph{witness} is a pair of such sets; $G$ \emph{is bipartite} if such a
  pair exists.
\end{definition}

\begin{definition}[Between (cut edges)]
  \label{def:between}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.between}
  \lean{TemporalGraph.Subgraph.between}
  \leanok
  The temporal graph \emph{between} two sets of vertices $s$ and $t$ is the
  restriction of $G$ to arcs that connect a vertex in $s$ to a vertex in $t$
  (in either direction). For a sub-temporal-graph $H$, $H.\mathrm{between}\ s\ t$
  is the same restriction of $H$, keeping the vertex set of $H$.
\end{definition}

\begin{definition}[Temporal bi-clique]
  \label{def:is-biclique}
  \uses{def:is-bipartite}
  \lean{TemporalGraph.IsBicliqueWith}
  \lean{TemporalGraph.IsBiclique}
  \lean{TemporalGraph.completeBipartite}
  \leanok
  A temporal graph $G$ is a \emph{temporal bi-clique} relative to disjoint sets
  $(s, t)$ if it is bipartite with witness $(s, t)$ and, moreover, every vertex
  in $s$ has an edge to every vertex in $t$ at some time. A set $S$ of vertices
  \emph{forms a bi-clique} in $G$ if $S = s \cup t$ for some pair $(s, t)$
  relative to which $G$ is a bi-clique; $G$ \emph{is a bi-clique} when $S$ is
  the set of all vertices. The
  \emph{complete bipartite} temporal graph on the disjoint union $\alpha \sqcup
  \beta$ with edge times given by a function $f : \alpha \to \beta \to
  \mathbb{N}$ is the temporal bi-clique where all such edges are present at
  their designated times. (SAND 2025, \S1.1: ``Temporal bi-cliques are defined
  analogously, except that the footprint is a complete bipartite graph.'')
\end{definition}

\begin{lemma}[Bipartiteness via simple footprint]
  \label{lem:is-bipartite-iff-footprint}
  \uses{def:is-bipartite,def:footprint}
  \lean{TemporalGraph.isBipartiteWith_iff_simpleFootprint}
  \leanok
  A symmetric, loopless temporal graph $G$ is bipartite with witness $(s,
  t)$ exactly when its simple footprint (the underlying undirected static
  graph) is bipartite with the same witness. This provides a bridge to
  Mathlib's \texttt{SimpleGraph.IsBipartiteWith}.
\end{lemma}

\begin{proof}
  \leanok
  Both sides say that $s$ and $t$ are disjoint and that every adjacent pair goes from one part to the other. Adjacency in the simple footprint is $\exists n,\ \mathrm{Adj}_G(x,y,n)$, so the two conditions unfold to the same statement.
\end{proof}

\begin{lemma}[A bi-clique part reaches the other]
  \label{lem:reachable-of-isBicliqueWith}
  \uses{def:is-biclique,def:reachable}
  \lean{TemporalGraph.Subgraph.reachable_of_isBicliqueWith}
  \leanok
  If $H$ is a bi-clique with witness $(s, t)$, every vertex of $s$ reaches
  every vertex of $t$ in $H$: they are joined directly by an edge.
\end{lemma}

\begin{proof}
  \leanok
  The bi-clique property gives an edge from $x$ to $y$ at some time $n$, which is a one-edge temporal walk.
\end{proof}

\begin{lemma}[Bi-cliques under induce and deleteVerts]
  \label{lem:isBicliqueWith-induce-deleteVerts}
  \uses{def:is-biclique,def:subgraph-induce-delete-add}
  \lean{TemporalGraph.Subgraph.isBicliqueWith_induce}
  \lean{TemporalGraph.Subgraph.isBicliqueWith_deleteVerts}
  \leanok
  Let $H$ be a bi-clique with witness $(s, t)$. For $s' \subseteq s$, the
  sub-temporal-graph induced by $H$ on $s' \cup t$ is a bi-clique with
  witness $(s', t)$ (SAND 2025, \S4.1, the restriction used by the one-sided
  split step). Dually, deleting any set $u$ of vertices from $H$ leaves a
  bi-clique with witness $(s \setminus u, t \setminus u)$.
\end{lemma}

\begin{proof}
  \leanok
  Disjointness restricts to subsets. An arc of the induced (resp.\ reduced) graph is an arc of $H$ with both endpoints kept, so it crosses between the parts, and the disjointness of $s$ and $t$ rules out the wrong orientation. The edges required between the kept parts are those of $H$, whose endpoints are kept.
\end{proof}
