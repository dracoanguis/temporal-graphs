\begin{definition}[Dismountable clique --- Theorem 2.1]
  \label{def:dismountable}
  \uses{def:sub-spanning-clique,def:earliest-latest-neighbor}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.Dismountable}
  A clique sub-temporal-graph $H$ of $G$ \emph{is dismountable} if it has a
  vertex that is simultaneously an earliest neighbor of some vertex and a
  latest neighbor of some (possibly different) vertex of $H$.
\end{definition}

\begin{definition}[$V^-$, $V^+$, $V^0$ --- def 3.1]
  \label{def:vminus-vplus-vzero}
  \uses{def:dismountable}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.Vminus}
  \lean{LooplessTemporalGraph.SubTemporalGraph.Vplus}
  \lean{LooplessTemporalGraph.SubTemporalGraph.Vzero}
  For $H$ a sub-temporal-graph of $G$: $V^-(H)$ is the set of vertices that
  are an earliest neighbor of some vertex of $H$; $V^+(H)$ is the set of
  vertices that are a latest neighbor of some vertex of $H$; and
  $V^0(H) = \mathrm{Verts}(H) \setminus (V^-(H) \cup V^+(H))$.
\end{definition}

\begin{lemma}[Partition of non-dismountable cliques --- lemma 3.1]
  \label{lem:partition-not-dismountable}
  \uses{def:dismountable,def:vminus-vplus-vzero}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.partition_of_not_dismountable}
  If $H$ is a clique that is not dismountable, then $V^-(H)$, $V^+(H)$,
  $V^0(H)$ partition $\mathrm{Verts}(H)$.
\end{lemma}

\begin{definition}[Two-hop dismountable]
  \label{def:two-hop-dismountable}
  \uses{def:dismountable}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.TwoHopDismountable}
  A variant of dismountability allowing the witnessing vertices to be
  identified up to a two-hop condition; every dismountable clique is
  two-hop dismountable (\lean{LooplessTemporalGraph.SubTemporalGraph.twoHopDismountable_of_dismountable}).
\end{definition}

\begin{lemma}[$V^+$/$V^-$ are not neighbors --- lemma 3.2]
  \label{lem:vplus-vminus-not-neighbors}
  \uses{def:two-hop-dismountable,def:vminus-vplus-vzero}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.vplus_ne_neighbor_of_not_twoHopHismountable}
  \lean{LooplessTemporalGraph.SubTemporalGraph.vminus_ne_neighbor_of_not_twoHopDismountable}
  If $H$ is not two-hop dismountable, no two distinct vertices of $V^+(H)$
  (resp.\ $V^-(H)$) are neighbors of each other via the earliest/latest edge
  structure witnessing their membership.
\end{lemma}

\begin{lemma}[$V^0$ is empty --- lemma 3.4]
  \label{lem:vzero-empty}
  \uses{lem:vplus-vminus-not-neighbors}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.vzero_empty_of_not_twoHopDismountable}
  If $H$ is a clique that is not two-hop dismountable, then $V^0(H) =
  \emptyset$, i.e.\ every vertex is an earliest or a latest neighbor of some
  vertex.
\end{lemma}

\begin{definition}[Dismount / mount]
  \label{def:dismount-mount}
  \uses{def:dismountable}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.dismount}
  \lean{LooplessTemporalGraph.SubTemporalGraph.mount}
  \lean{LooplessTemporalGraph.SubTemporalGraph.twoHopMount}
  $H.\mathrm{dismount}(x)$ removes vertex $x$ from $H$ (a special case of
  \texttt{deleteVerts}). $H.\mathrm{mount}(x, \dots)$ (resp.\
  $\mathrm{twoHopMount}$) is the corresponding reconstruction operation used
  to transfer a spanner of the dismounted clique back to $H$.
\end{definition}

\begin{theorem}[Dismount/mount preserves spanning correctness]
  \label{thm:dismountability}
  \uses{def:dismount-mount,def:spanner}
  % TODO post-rename
  \lean{LooplessTemporalGraph.SubTemporalGraph.mount_tc}
  \lean{LooplessTemporalGraph.SubTemporalGraph.dismountability}
  \lean{LooplessTemporalGraph.SubTemporalGraph.twoHopDismountability}
  If $H'$ is a TC-spanner of $H.\mathrm{dismount}(x)$, then mounting $x$
  back onto $H'$ yields a TC-spanner of $H$ (and similarly for
  \texttt{twoHopMount}).
\end{theorem}
