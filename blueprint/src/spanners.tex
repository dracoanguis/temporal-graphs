\chapter{Sub-temporal-graphs and spanners}
\label{chap:spanners}

Formalized in \texttt{TemporalGraph/Subgraph.lean} (general subgraph theory)
and \texttt{TemporalGraph/Connectivity/Subgraph.lean} (reachability and
spanners at the subgraph level).

\begin{definition}[Sub-temporal-graph]
  \label{def:sub-temporal-graph}
  \uses{def:temporal-graph}
  \lean{TemporalGraph.Subgraph}
  \leanok
  A \emph{sub-temporal-graph} $H$ of a temporal graph $G$ is a choice of
  vertex subset $\mathrm{Verts}(H) \subseteq V$ together with an edge
  relation on it that is compatible with (a restriction of) $G$'s edges. In
  the library, $H$ literally \emph{is} a temporal graph on $V$ (extending
  Definition~\ref{def:temporal-graph}) together with $\mathrm{Verts}(H)$ and
  proofs that every one of its edges is one of $G$'s and has both endpoints
  in $\mathrm{Verts}(H)$; in particular $H$ need not be symmetric even when
  $G$ is.
\end{definition}

\begin{definition}[Spanning, clique sub-temporal-graph]
  \label{def:sub-spanning-clique}
  \uses{def:sub-temporal-graph,def:is-simple-clique}
  \lean{TemporalGraph.Subgraph.IsSpanning}
  \lean{TemporalGraph.Subgraph.IsCliqueSubgraph}
  \leanok
  $H$ \emph{is spanning} in $G$ if $\mathrm{Verts}(H) = V$. $H$ \emph{is a
  clique} (or a \emph{clique subgraph} via \texttt{IsCliqueSubgraph}) if its vertex set $\mathrm{Verts}(H)$ forms a clique in $H$ itself
  (Definition~\ref{def:is-simple-clique}): every pair of distinct vertices of
  $\mathrm{Verts}(H)$ is joined by an edge of $H$ at some time.
\end{definition}

\begin{definition}[Order on sub-temporal-graphs]
  \label{def:subgraph-order}
  \uses{def:sub-temporal-graph}
  \lean{TemporalGraph.Subgraph.top_toTemporalGraph}
  \leanok
  Sub-temporal-graphs of $G$ are ordered by inclusion: $H \le H'$ when
  $\mathrm{Verts}(H) \subseteq \mathrm{Verts}(H')$ and every edge of $H$ is
  one of $H'$ — the analogue, one level down, of
  Definition~\ref{def:spanning-order}. This makes the sub-temporal-graphs of
  $G$ a bounded partial order: the bottom element $\bot$ is the empty
  sub-temporal-graph, and the top element $\top$ is $G$ itself, seen as a
  sub-temporal-graph of itself on all of $V$.
\end{definition}

\begin{lemma}[Order via edge sets and vertex sets]
  \label{lem:subgraph-le-iff-edgeSet-verts}
  \uses{def:subgraph-order,def:arcset-edgeset}
  \lean{TemporalGraph.Subgraph.le_iff_edgeSet_and_verts}
  \leanok
  When $H'$ is symmetric, $H \le H'$ exactly when
  $\mathrm{EdgeSet}(H) \subseteq \mathrm{EdgeSet}(H')$ and
  $\mathrm{Verts}(H) \subseteq \mathrm{Verts}(H')$.
\end{lemma}

\begin{lemma}[$\le$ is compatible with \texttt{toTemporalGraph}]
  \label{lem:toTemporalGraph-le}
  \uses{def:subgraph-order}
  \lean{TemporalGraph.Subgraph.toTemporalGraph_le}
  \lean{TemporalGraph.Subgraph.toTemporalGraph_le_self}
  \leanok
  If $H \le H'$ as sub-temporal-graphs of $G$, then
  $H.\mathrm{toTemporalGraph} \le H'.\mathrm{toTemporalGraph}$, the inherited
  order of Definition~\ref{def:spanning-order}. Every sub-temporal-graph's
  $\mathrm{toTemporalGraph}$ is, in particular, a spanning sub-temporal-graph
  of $G$ itself.
\end{lemma}

\begin{definition}[Union of sub-temporal-graphs]
  \label{def:subgraph-sup}
  \uses{def:subgraph-order,def:arcset-edgeset}
  \lean{TemporalGraph.Subgraph.sup_edgeSet}
  \leanok
  Sub-temporal-graphs of $G$ form a \texttt{SemilatticeSup}: $H_1 \sqcup H_2$
  has vertex set $\mathrm{Verts}(H_1) \cup \mathrm{Verts}(H_2)$ and keeps an
  edge of either $H_1$ or $H_2$, and it is the least upper bound of $H_1$
  and $H_2$ for the order of Definition~\ref{def:subgraph-order}. Its edge
  set is the union $\mathrm{EdgeSet}(H_1) \cup \mathrm{EdgeSet}(H_2)$.
\end{definition}

\begin{definition}[Induced, added, deleted sub-temporal-graphs]
  \label{def:subgraph-induce-delete-add}
  \uses{def:sub-temporal-graph}
  \lean{TemporalGraph.Subgraph.IsInduced}
  \lean{TemporalGraph.Subgraph.induce}
  \lean{TemporalGraph.Subgraph.deleteVerts}
  \lean{TemporalGraph.Subgraph.addVerts}
  \leanok
  $H$ \emph{is induced} if every $G$-edge between two vertices of $H$ is
  already an edge of $H$. Given $s \subseteq V$, the sub-temporal-graph
  $H[s]$ \emph{induced by $H$ on $s$} has vertex set $s$ and keeps exactly
  the edges of $H$ whose both endpoints lie in $s$; it is always a
  sub-temporal-graph of $G$, whether or not $s \subseteq \mathrm{Verts}(H)$.
  Deleting $s$ from $H$ is $H[\mathrm{Verts}(H) \setminus s]$; adding $s$ to
  $H$ is $H[\mathrm{Verts}(H) \cup s]$ — both special cases of induction,
  and both below, resp.\ above, $H$ in the order of
  Definition~\ref{def:subgraph-order}.
\end{definition}

\begin{lemma}[Finiteness is preserved by induction and deletion]
  \label{lem:subgraph-finite-induce-delete}
  \uses{def:subgraph-induce-delete-add}
  \lean{TemporalGraph.Subgraph.finite_induce_verts}
  \lean{TemporalGraph.Subgraph.deleteVerts_finite}
  \leanok
  If $\mathrm{Verts}(H)$ is finite, then so is $\mathrm{Verts}(H[s])$ for
  any $s \subseteq \mathrm{Verts}(H)$ (in particular, so is
  $\mathrm{Verts}(H \setminus s)$, as an instance named
  \texttt{deleteVerts\_finite}): both vertex sets are subsets of
  $\mathrm{Verts}(H)$.
\end{lemma}

\begin{lemma}[Finitely many vertices, simple: finitely many edges]
  \label{lem:subgraph-finite-edgeSet-of-simple}
  \uses{def:sub-spanning-clique,def:arcset-edgeset}
  \lean{TemporalGraph.Subgraph.finite_edgeSet_of_finite_simple}
  \leanok
  If $H$ is simple (Definition~\ref{def:is-simple-clique}) and
  $\mathrm{Verts}(H)$ is finite, then $\mathrm{EdgeSet}(H)$ is finite: there is
  at most one time per unordered pair of vertices, and there are finitely
  many such pairs.
\end{lemma}

\begin{definition}[Adding a single edge]
  \label{def:subgraph-addEdge}
  \uses{def:sub-temporal-graph}
  \lean{TemporalGraph.Subgraph.addArc}
  \lean{TemporalGraph.Subgraph.addEdge}
  \leanok
  Given an edge $(x,y,n)$ of $G$ with $x, y \in \mathrm{Verts}(H)$: adding it
  to $H$ in one direction ($\mathrm{addArc}$) needs no hypothesis on $G$ or
  $H$ at all, and inserts only the arc $x \to y$ at time $n$. Adding it in
  both directions at once ($\mathrm{addEdge}$) needs $G$ symmetric, to
  find the reverse arc $y \to x$ back in $G$.
\end{definition}

\begin{lemma}[Adding an edge changes the edge set by one]
  \label{lem:subgraph-addEdge-edgeSet}
  \uses{def:subgraph-addEdge,def:arcset-edgeset}
  \lean{TemporalGraph.Subgraph.addEdge_edgeSet}
  \lean{TemporalGraph.Subgraph.ncard_edgeSet_addEdge_le}
  \leanok
  $H.\mathrm{addEdge}$'s edge set is $\mathrm{EdgeSet}(H)$ with the new pair
  $(\{x,y\}, n)$ inserted; in particular its \texttt{Set.ncard} is at most
  $|\mathrm{EdgeSet}(H)| + 1$, with no finiteness hypothesis needed.
\end{lemma}

\begin{lemma}[Earliest / latest edge from a finite incidence set]
  \label{lem:earliest-latest-edge-exists}
  \uses{def:incidence-set,def:earliest-latest-neighbor}
  \lean{TemporalGraph.Subgraph.earliestEdge_exists_of_nonempty_finite_incidenceSet}
  \lean{TemporalGraph.Subgraph.latestEdge_exists_of_nonempty_finite_incidenceSet}
  \leanok
  If $x$'s incidence set in $H$ is nonempty and finite, $x$ has an earliest
  edge in $H$; dually, it has a latest edge. (A nonempty finite set of times
  has a minimum and a maximum.) Combined with
  Definition~\ref{def:sub-spanning-clique}, a finite nontrivial simple
  clique gives every vertex both an earliest and a latest edge — the
  starting point of Chapter~\ref{chap:dismountability}.
\end{lemma}

\begin{definition}[Footprint edge set]
  \label{def:subgraph-footprint-edgeSet}
  \uses{def:sub-temporal-graph}
  \lean{TemporalGraph.Subgraph.footprintEdgeSet}
  \leanok
  The \emph{footprint edge set} of $H$ forgets time directly into unordered
  pairs: it is the set of $\{x,y\} \in \mathrm{Sym2}\,V$ such that
  $H.\mathrm{Adj}(x,y,n)$ for some $n$ — the same forgetful step as
  Definition~\ref{def:footprint}'s footprint digraph, but landing on
  $\mathrm{Sym2}\,V$ (matching the shape of $\mathrm{EdgeSet}$ without the
  time component) rather than on a directed graph.
\end{definition}

\begin{lemma}[Edges and footprint edges are in bijection]
  \label{lem:edgeSet-footprint-bijective}
  \uses{def:subgraph-footprint-edgeSet,def:arcset-edgeset,def:is-simple-clique}
  \lean{TemporalGraph.Subgraph.edgeSetToFootprintEdgeSet_bijective}
  \leanok
  If $H$ is symmetric and simple, the map from $\mathrm{EdgeSet}(H)$ to
  $H$'s footprint edge set, forgetting the time component, is a bijection:
  simplicity guarantees each unordered pair of $H$-adjacent vertices carries
  a single time. In particular the \texttt{Set.ncard} of $\mathrm{EdgeSet}(H)$
  agrees with that of the footprint edge set in this case, with no
  finiteness hypothesis needed
  (\lean{TemporalGraph.Subgraph.ncard_edgeSet_of_simple}), and so does
  $H.\mathrm{size}$ under a \texttt{Fintype} instance
  (\lean{TemporalGraph.Subgraph.size_of_simple}).
\end{lemma}

\begin{definition}[Matching]
  \label{def:subgraph-matching}
  \uses{def:sub-temporal-graph,def:sub-spanning-clique}
  \lean{TemporalGraph.Subgraph.IsMatching}
  \lean{TemporalGraph.Subgraph.IsPerfectMatching}
  \leanok
  $M$ \emph{is a matching} if every vertex of $\mathrm{Verts}(M)$ has exactly
  one outgoing $M$-arc (a unique pair $(y,n)$ with $M.\mathrm{Adj}(x,y,n)$).
  $M$ \emph{is a perfect matching} if it is a matching and is spanning.
\end{definition}

\begin{definition}[Building a sub-temporal-graph]
  \label{def:subgraph-toSubgraph-fromEdgeSet}
  \uses{def:sub-temporal-graph,def:spanning-order,def:arcset-edgeset}
  \lean{TemporalGraph.toSubgraph}
  \lean{TemporalGraph.Subgraph.fromEdgeSet}
  \leanok
  Any temporal graph $H$ with $H \le G$ (Definition~\ref{def:spanning-order})
  gives a spanning sub-temporal-graph of $G$ with the same adjacency,
  licensed by $H \le G$. Dually, a set of unordered edges yields a
  sub-temporal-graph of $G$: its vertices are those touched by an edge of
  the set, and its adjacencies are exactly the members of the set that
  already are edges of $G$ — the sub-temporal-graph analogue of
  Definition~\ref{def:from-arc-edge-set}'s $\mathrm{fromEdgeSet}$.
\end{definition}

\begin{definition}[Reachability and spanning for sub-temporal-graphs]
  \label{def:subgraph-reachable-tc-spannerOf}
  \uses{def:sub-temporal-graph,def:reachable}
  \lean{TemporalGraph.Subgraph.IsSpannerOf}
  \leanok
  On a sub-temporal-graph $H$, reachability and temporal connectivity are
  inherited from the parent $G$ — they operate on vertices of type $V$. Given
  two sub-temporal-graphs $H, H'$ of $G$: $H$ \emph{is a spanner of} $H'$
  if $H \le H'$, they share the same vertex set, and every pair of vertices
  in that set that is reachable in $H'$ remains reachable in $H$ — the
  relation used by the dismount/mount operations of
  Chapter~\ref{chap:dismountability} to transfer a spanner of a smaller
  clique back onto the original one.
\end{definition}

\begin{lemma}[A reachable pair of distinct vertices lies in $H$]
  \label{lem:subgraph-mem-verts-of-reachable}
  \uses{def:sub-temporal-graph,def:reachable}
  \lean{TemporalGraph.Subgraph.mem_verts_of_reachable}
  \leanok
  If $x \neq y$ and $y$ is reachable from $x$ in $H$, then both
  $x, y \in \mathrm{Verts}(H)$: a nonempty witnessing walk has a first and a
  last edge, both of $H$, giving each endpoint's membership.
\end{lemma}

\begin{definition}[Temporal spanner]
  \label{def:spanner}
  \uses{def:sub-spanning-clique,def:reachable,def:subgraph-reachable-tc-spannerOf}
  \lean{TemporalGraph.Subgraph.IsSpanner}
  \lean{TemporalGraph.Subgraph.IsTCSpanner}
  \leanok
  A spanning sub-temporal-graph $H$ of $G$ \emph{is a spanner} if, for all
  $x, y \in \mathrm{Verts}(H)$, whenever $x$ reaches $y$ in $G$, then $x$ reaches
  $y$ in $H$. $H$ \emph{is a TC-spanner} if it is spanning and $\mathrm{Verts}(H)$ is
  temporally connected in $H$.
\end{definition}

\begin{lemma}[TC-spanner exists iff temporally connected]
  \label{lem:tcspanner-iff-tc}
  \uses{def:spanner}
  \lean{TemporalGraph.Subgraph.TCspanner_iff_TC}
  \leanok
  $G$ admits a TC-spanner if and only if $G$ is temporally connected.
\end{lemma}
